cmake_minimum_required(VERSION 3.20)
project(okam_windows_probe C)

add_executable(okam-windows-probe p2p_probe.c)
target_compile_definitions(okam-windows-probe PRIVATE WIN32_LEAN_AND_MEAN)
add_executable(okam-devdll-probe devdll_probe.c)
target_compile_definitions(okam-devdll-probe PRIVATE WIN32_LEAN_AND_MEAN)
target_link_libraries(okam-devdll-probe PRIVATE advapi32)
add_executable(okam-stream-probe stream_probe.c)
target_compile_definitions(okam-stream-probe PRIVATE WIN32_LEAN_AND_MEAN)
target_link_libraries(okam-stream-probe PRIVATE advapi32)
if(MINGW)
  target_link_options(okam-windows-probe PRIVATE -static)
  target_link_options(okam-devdll-probe PRIVATE -static)
  target_link_options(okam-stream-probe PRIVATE -static)
endif()
